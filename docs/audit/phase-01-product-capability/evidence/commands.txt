## 6. Persistence signal sweep (Supabase touch-point census per src/lib module)
$ grep -rl "createSupabaseServerClient\|createServerClient\|supabaseAdmin\|@supabase" src/lib --include="*.ts" | wc -l
# => initial pass (17) undercounted; corrected pattern below is authoritative
$ for d in src/lib/*/; do
    name=$(basename "$d")
    cnt=$(grep -rl "createSupabaseServerClient\|createServerClient\|supabaseAdmin" "$d" --include="*.ts" 2>/dev/null | wc -l)
    total=$(find "$d" -name "*.ts" -not -name "*.test.ts" | wc -l)
    echo "$name: $cnt/$total files touch supabase"
  done
# => full per-module output preserved in 03-product-module-inventory.md and 05-capability-evidence.md.
# CAVEAT (discovered mid-audit, corrected): this pattern under-detects modules that go through a named
# wrapper, e.g. `createSupabaseServiceRoleClient` (src/lib/founder-program/db.ts:1) or
# `createSupabaseBrowserClient` (src/lib/workspace/runtime-persistence.ts:1). Domain agents were
# instructed to grep the broader set {createSupabaseServerClient, createSupabaseServiceRoleClient,
# createSupabaseBrowserClient, .from(} and to trace one import level deeper before concluding
# "no persistence" for any module that shows zero direct hits — to distinguish legitimate
# aggregation-only layering (module reads through another module that does persist) from a genuinely
# unpersisted/simulated capability.
